    <script>
        function buildDenseBlossoms() {
            const container = document.getElementById('dense-blossoms');
            if (!container) return;

            // Classic heart equation with sharp top cleft and pronounced bottom point: (x^2 + y^2 - 1)^3 - x^2 * y^3 <= 0
            // Centered lower down around (250, 260) with scaling factor 175
            function isInHeart(x, y) {
                const nx = (x - 250) / 175;
                const ny = -(y - 260) / 175;
                const a = nx * nx + ny * ny - 1;
                return (a * a * a - nx * nx * ny * ny * ny) <= 0;
            }

            let blossomsHTML = '';
            // Fine-grained grid scan to outline the classic heart silhouette clearly
            const step = 14;
            for (let x = 50; x <= 450; x += step) {
                for (let y = 60; y <= 420; y += step) {
                    // Slight organic position jitter so it doesn't look like a rigid grid
                    const offsetX = (Math.random() - 0.5) * 11;
                    const offsetY = (Math.random() - 0.5) * 11;
                    const px = x + offsetX;
                    const py = y + offsetY;

                    // Probability check to leave subtle natural gaps inside the crown
                    if (isInHeart(px, py) && Math.random() > 0.22) {
                        const fontSize = Math.floor(Math.random() * 8) + 26; // Visible flower size
                        const opacity = (Math.random() * 0.15 + 0.85).toFixed(2);
                        const rotation = Math.floor(Math.random() * 360); // Random organic angle
                        blossomsHTML += `<text x="${px.toFixed(1)}" y="${py.toFixed(1)}" font-size="${fontSize}" opacity="${opacity}" transform="rotate(${rotation}, ${px.toFixed(1)}, ${py.toFixed(1)})" text-anchor="middle" dominant-baseline="central" style="user-select:none;">🌸</text>`;
                    }
                }
            }
            container.innerHTML = blossomsHTML;
        }

        buildDenseBlossoms();

        let startDate = localStorage.getItem('startDate') || new Date(Date.now() - 365 * 24 * 60 * 60 * 1000).toISOString();

        function updateCounter() {
            const start = new Date(startDate);
            const now = new Date();
            let diff = now - start;

            if (diff < 0) diff = 0;

            const secondsTotal = Math.floor(diff / 1000);
            const seconds = secondsTotal % 60;

            document.getElementById('seconds').innerText = String(seconds).padStart(2, '0');
        }

        setInterval(updateCounter, 1000);
        updateCounter();

        const fallingContainer = document.getElementById('falling-container');
        const symbols = ['🌸', '💖', '💗', '✨', '🌸', '🌷', '💕', '🌸'];

        function createFallingItem() {
            const item = document.createElement('div');
            item.className = 'falling-item';
            item.innerText = symbols[Math.floor(Math.random() * symbols.length)];
            
            const startLeft = Math.random() * 100;
            const duration = Math.random() * 4 + 4;
            const size = Math.random() * 1.5 + 1.2;

            item.style.left = startLeft + 'vw';
            item.style.fontSize = size + 'rem';
            item.style.animationDuration = duration + 's';
            
            fallingContainer.appendChild(item);

            setTimeout(() => {
                item.remove();
            }, duration * 1000);
        }

        setInterval(createFallingItem, 300);
